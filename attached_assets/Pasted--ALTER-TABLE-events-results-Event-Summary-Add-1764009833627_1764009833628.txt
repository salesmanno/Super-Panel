-- ============================================
-- ALTER TABLE events_results - Event Summary
-- Adds comprehensive income/expense tracking
-- ============================================

ALTER TABLE events_results
-- Revenue: Ticket sales breakdown
ADD COLUMN revenue_tickets_cash DECIMAL(10,2) DEFAULT 0.00 COMMENT 'Billettinntekter - Kontant' AFTER total_ticket_sales,
ADD COLUMN revenue_tickets_card DECIMAL(10,2) DEFAULT 0.00 COMMENT 'Billettinntekter - Kort' AFTER revenue_tickets_cash,
ADD COLUMN revenue_tickets_vipps DECIMAL(10,2) DEFAULT 0.00 COMMENT 'Billettinntekter - Vipps' AFTER revenue_tickets_card,
ADD COLUMN revenue_tickets_invoice DECIMAL(10,2) DEFAULT 0.00 COMMENT 'Billettinntekter - Faktura' AFTER revenue_tickets_vipps,

-- Revenue: Additional sources
ADD COLUMN revenue_merchandise DECIMAL(10,2) DEFAULT 0.00 COMMENT 'Salg av varer/merchandise' AFTER revenue_tickets_invoice,
ADD COLUMN revenue_donations DECIMAL(10,2) DEFAULT 0.00 COMMENT 'Donasjoner' AFTER revenue_merchandise,

-- Expenses: Cost breakdown
ADD COLUMN expense_staff DECIMAL(10,2) DEFAULT 0.00 COMMENT 'Personalkostnader' AFTER total_costs,
ADD COLUMN expense_venue DECIMAL(10,2) DEFAULT 0.00 COMMENT 'Venueleie og lokalekostand' AFTER expense_staff,
ADD COLUMN expense_equipment DECIMAL(10,2) DEFAULT 0.00 COMMENT 'Utleie av utstyr' AFTER expense_venue,
ADD COLUMN expense_materials DECIMAL(10,2) DEFAULT 0.00 COMMENT 'Materialer og forbruk' AFTER expense_equipment,
ADD COLUMN expense_other DECIMAL(10,2) DEFAULT 0.00 COMMENT 'Andre utgifter' AFTER expense_materials,

-- Attendance & Engagement
ADD COLUMN no_shows INT(11) DEFAULT 0 COMMENT 'Antall som ikke møtte opp' AFTER total_checkins,
ADD COLUMN attendance_rate DECIMAL(5,2) DEFAULT 0.00 COMMENT 'Oppmøte % (0-100)' AFTER no_shows,
ADD COLUMN average_age INT(11) DEFAULT 0 COMMENT 'Gjennomsnittlig alder' AFTER attendance_rate,

-- Event Status & Notes
ADD COLUMN status VARCHAR(50) DEFAULT 'planning' COMMENT 'Status: planning, confirmed, in_progress, completed, cancelled' AFTER event_cancelled,
ADD COLUMN notes LONGTEXT DEFAULT NULL COMMENT 'Notater om arrangementet' AFTER status,
ADD COLUMN organizer_notes LONGTEXT DEFAULT NULL COMMENT 'Interne notater' AFTER notes,

-- Timestamps
ADD COLUMN finalized_at DATETIME DEFAULT NULL COMMENT 'Når resultatene ble avsluttet' AFTER updated_at;
