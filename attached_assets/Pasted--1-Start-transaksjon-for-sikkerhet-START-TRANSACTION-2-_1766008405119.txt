ALTER TABLE `reservation_services`

FOREIGN KEY (`created_by`) REFERENCES `tenant_admins`(`id`) ON DELETE SET NULL;

-- 6. Lag DE NYE tabellene kun hvis de ikke finnes
-- service_bookings
CREATE TABLE IF NOT EXISTS `service_bookings` (
  `id` CHAR(36) NOT NULL DEFAULT UUID(),
  `company_id` CHAR(36) NOT NULL,
  `service_id` CHAR(36) NOT NULL,
  `customer_id` CHAR(36) NOT NULL,
  `staff_id` CHAR(36) DEFAULT NULL,
  `booking_date` DATE NOT NULL,
  `start_time` TIME NOT NULL,
  `end_time` TIME NOT NULL,
  `status` VARCHAR(20) DEFAULT 'confirmed',
  `payment_status` VARCHAR(20) DEFAULT 'pending',
  `payment_method` VARCHAR(50) DEFAULT NULL,
  `notes` TEXT DEFAULT NULL,
  `created_at` DATETIME DEFAULT CURRENT_TIMESTAMP(),
  `updated_at` DATETIME DEFAULT CURRENT_TIMESTAMP() ON UPDATE CURRENT_TIMESTAMP(),
  
  PRIMARY KEY (`id`),
  UNIQUE KEY `unique_booking_time` (`company_id`, `booking_date`, `start_time`, `staff_id`),
  KEY `idx_service_bookings_staff_date` (`staff_id`, `booking_date`),
  KEY `idx_service_bookings_company` (`company_id`),
  KEY `idx_service_bookings_service` (`service_id`),
  KEY `idx_service_bookings_customer` (`customer_id`),
  KEY `idx_service_bookings_status` (`status`),
  
  FOREIGN KEY (`company_id`) REFERENCES `tenant_companies`(`id`) ON DELETE CASCADE,
  FOREIGN KEY (`service_id`) REFERENCES `reservation_services`(`id`) ON DELETE CASCADE,
  FOREIGN KEY (`customer_id`) REFERENCES `tenant_customers`(`id`) ON DELETE CASCADE,
  FOREIGN KEY (`staff_id`) REFERENCES `tenant_staff`(`id`) ON DELETE SET NULL
);

-- staff_availability
CREATE TABLE IF NOT EXISTS `staff_availability` (
  `id` CHAR(36) NOT NULL DEFAULT UUID(),
  `company_id` CHAR(36) NOT NULL,
  `staff_id` CHAR(36) NOT NULL,
  
  `date` DATE NOT NULL,
  `day_of_week` TINYINT(1) DEFAULT NULL COMMENT '0=Søndag, 1=Mandag, etc.',
  `start_time` TIME DEFAULT '09:00:00',
  `end_time` TIME DEFAULT '17:00:00',
  `is_working_day` TINYINT(1) DEFAULT 1,
  
  `booked_slots` TEXT DEFAULT NULL,
  `breaks` TEXT DEFAULT NULL,
  
  `max_bookings_per_day` INT(11) DEFAULT NULL,
  `slot_duration_minutes` INT(11) DEFAULT 60,
  `buffer_between_minutes` INT(11) DEFAULT 0,
  
  `availability_type` VARCHAR(20) DEFAULT 'standard',
  `notes` TEXT DEFAULT NULL,
  
  `created_at` DATETIME DEFAULT CURRENT_TIMESTAMP(),
  `updated_at` DATETIME DEFAULT CURRENT_TIMESTAMP() ON UPDATE CURRENT_TIMESTAMP(),
  
  PRIMARY KEY (`id`),
  UNIQUE KEY `unique_staff_date` (`company_id`, `staff_id`, `date`),
  KEY `idx_staff_availability_staff` (`staff_id`),
  KEY `idx_staff_availability_date_range` (`date`, `start_time`, `end_time`),
  KEY `idx_staff_availability_type` (`availability_type`),
  
  FOREIGN KEY (`company_id`) REFERENCES `tenant_companies`(`id`) ON DELETE CASCADE,
  FOREIGN KEY (`staff_id`) REFERENCES `tenant_staff`(`id`) ON DELETE CASCADE
);

-- staff_services
CREATE TABLE IF NOT EXISTS `staff_services` (
  `id` CHAR(36) NOT NULL DEFAULT UUID(),
  `company_id` CHAR(36) NOT NULL,
  `staff_id` CHAR(36) NOT NULL,
  `service_id` CHAR(36) NOT NULL,
  `is_active` BOOLEAN DEFAULT TRUE,
  `hourly_rate` DECIMAL(10,2) DEFAULT NULL,
  `max_daily_bookings` INT(11) DEFAULT 5,
  `created_at` DATETIME DEFAULT CURRENT_TIMESTAMP(),
  `updated_at` DATETIME DEFAULT CURRENT_TIMESTAMP() ON UPDATE CURRENT_TIMESTAMP(),
  
  PRIMARY KEY (`id`),
  UNIQUE KEY `unique_staff_service` (`company_id`, `staff_id`, `service_id`),
  
  FOREIGN KEY (`company_id`) REFERENCES `tenant_companies`(`id`) ON DELETE CASCADE,
  FOREIGN KEY (`staff_id`) REFERENCES `tenant_staff`(`id`) ON DELETE CASCADE,
  FOREIGN KEY (`service_id`) REFERENCES `reservation_services`(`id`) ON DELETE CASCADE
);

-- 7. Reaktiver foreign key checks
SET FOREIGN_KEY_CHECKS = 1;

-- 8. Commit transaksjonen
COMMIT;
