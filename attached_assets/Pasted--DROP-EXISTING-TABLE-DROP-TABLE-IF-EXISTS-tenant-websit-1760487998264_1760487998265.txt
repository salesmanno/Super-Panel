-- ==================== DROP EXISTING TABLE ====================
DROP TABLE IF EXISTS tenant_website_config;

-- ==================== TENANT WEBSITE CONFIGURATION ====================
CREATE TABLE tenant_website_config (
    id CHAR(36) PRIMARY KEY DEFAULT (UUID()),
    company_id CHAR(36) NOT NULL,
    module_id CHAR(36) NOT NULL,
    
    -- Module visibility control
    is_enabled BOOLEAN DEFAULT TRUE,
    is_public BOOLEAN DEFAULT TRUE,
    show_in_navigation BOOLEAN DEFAULT TRUE,
    
    -- Display settings
    display_name VARCHAR(255),
    display_order INTEGER DEFAULT 0,
    navigation_icon VARCHAR(100),
    
    -- Content configuration
    custom_content TEXT, -- For custom HTML/content
    about_us_content TEXT, -- Specific field for about us content
    
    -- Styling and layout
    template_data TEXT,
    style_config TEXT,
    layout_config TEXT,
    color_scheme VARCHAR(50) DEFAULT 'default',
    
    -- SEO configuration
    seo_title VARCHAR(255),
    seo_description TEXT,
    seo_keywords TEXT,
    canonical_url VARCHAR(500),
    
    -- Access control
    allowed_roles TEXT, -- JSON array of roles that can access if not public
    access_password VARCHAR(255), -- Optional password protection
    
    -- Status and publishing
    is_published BOOLEAN DEFAULT TRUE,
    published_at DATETIME,
    unpublished_at DATETIME,
    
    created_by CHAR(36) NOT NULL,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    updated_by CHAR(36),
    
    FOREIGN KEY (company_id) REFERENCES tenant_companies(id) ON DELETE CASCADE,
    FOREIGN KEY (module_id) REFERENCES system_modules(id) ON DELETE CASCADE,
    FOREIGN KEY (created_by) REFERENCES saas_admins(id), -- SaaS admin controls website config
    FOREIGN KEY (updated_by) REFERENCES saas_admins(id),
    UNIQUE(company_id, module_id)
);

-- ==================== TENANT GENERAL WEBSITE SETTINGS ====================
CREATE TABLE tenant_website_general (
    id CHAR(36) PRIMARY KEY DEFAULT (UUID()),
    company_id CHAR(36) NOT NULL,
    
    -- Branding
    site_title VARCHAR(255),
    site_description TEXT,
    logo_url VARCHAR(500),
    favicon_url VARCHAR(500),
    primary_color VARCHAR(7) DEFAULT '#3366CC',
    secondary_color VARCHAR(7) DEFAULT '#666666',
    
    -- Layout
    header_layout VARCHAR(50) DEFAULT 'standard',
    footer_layout VARCHAR(50) DEFAULT 'standard',
    navigation_style VARCHAR(50) DEFAULT 'top',
    
    -- Homepage settings
    homepage_module CHAR(36), -- Which module to show on homepage
    hero_enabled BOOLEAN DEFAULT TRUE,
    hero_title VARCHAR(255),
    hero_description TEXT,
    hero_image_url VARCHAR(500),
    hero_button_text VARCHAR(100),
    hero_button_url VARCHAR(500),
    
    -- Contact information
    contact_email VARCHAR(255),
    contact_phone VARCHAR(50),
    contact_address TEXT,
    social_media_links TEXT, -- JSON with social media URLs
    
    -- Footer content
    footer_text TEXT,
    footer_links TEXT, -- JSON array of footer links
    
    -- Advanced settings
    custom_css TEXT,
    custom_js TEXT,
    google_analytics_id VARCHAR(100),
    
    -- Status
    is_active BOOLEAN DEFAULT TRUE,
    maintenance_mode BOOLEAN DEFAULT FALSE,
    maintenance_message TEXT,
    
    created_by CHAR(36) NOT NULL,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    updated_by CHAR(36),
    
    FOREIGN KEY (company_id) REFERENCES tenant_companies(id) ON DELETE CASCADE,
    FOREIGN KEY (homepage_module) REFERENCES system_modules(id),
    FOREIGN KEY (created_by) REFERENCES saas_admins(id),
    FOREIGN KEY (updated_by) REFERENCES saas_admins(id),
    UNIQUE(company_id)
);

-- ==================== INDEXES ====================
CREATE INDEX idx_website_config_company ON tenant_website_config(company_id);
CREATE INDEX idx_website_config_public ON tenant_website_config(is_public, is_enabled, is_published);
CREATE INDEX idx_website_config_navigation ON tenant_website_config(show_in_navigation, display_order);
